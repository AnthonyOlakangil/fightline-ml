# Minimal runtime for the deployed demo.
#
# The app loads precomputed artefacts and never trains, scrapes, or re-scores, so it
# needs far less than the research project: no duckdb, matplotlib, typer, requests,
# beautifulsoup4 or scikit-learn. lightgbm loads the saved model's text to score a
# hypothetical matchup; the calibrator is a table of thresholds, applied with numpy.
#
# Versions are pinned to exactly what uv.lock resolves: the model is a pickle, and an
# unpickle across scikit-learn/LightGBM versions is not guaranteed to load or to score
# identically. tests/test_app.py fails if these drift from the lock.
streamlit==1.65.0
pandas==3.0.6
pyarrow==25.0.1
numpy==2.4.6
lightgbm==4.7.0
